Starter stylesheet (only for a student with no Week 8 work)

styles.css

/* My resume: one stylesheet for all my pages */

/* Text: a font list ending in a generic family, space between lines, dark grey text */
body {
  font-family: Verdana, Tahoma, sans-serif;
  line-height: 1.5;
  color: #222222;
}

/* Element selector: every h1, a gradient header with white text (8.8:1 or more) */
h1 {
  font-family: Georgia, "Times New Roman", serif;
  color: #ffffff;
  background-image: linear-gradient(to right, #13305c, #8a1c55);
  text-align: center;
}

/* Class selector: every element with class="highlight" */
.highlight {
  background-color: #fff3c4;
}

/* ID selector: only the element with id="about" */
#about {
  font-style: italic;
}

/* Box model: room and a line under every section heading */
h2 {
  border-bottom: 2px solid #13305c;
  padding-bottom: 0.25em;
  margin-top: 1.5em;
}

/* Lists: the nav list with no bullets and no indent, its items on one line */
nav ul {
  list-style-type: none;
  margin: 0;
  padding: 0;
}
nav li {
  margin-right: 0.5em;
  display: inline-block;
}

/* Lists: square bullets for my skills */
.skills {
  list-style-type: square;
}

/* Links in nav, in state order: white on navy (13.0:1), maroon under the mouse or the keyboard (8.8:1) */
nav a:link, nav a:visited {
  color: #ffffff;
  background-color: #13305c;
  padding: 0.25em 0.75em;
  text-decoration: none;
}
nav a:hover, nav a:focus {
  background-color: #8a1c55;
}

/* Position: my h1 is the reference box; the ribbon sits in its top-right corner, above it (14.2:1) */
h1 {
  position: relative;
}
.ribbon {
  position: absolute;
  top: 8px;
  right: 8px;
  z-index: 1;
  padding: 0 0.5em;
  font-size: 12px;
  color: #222222;
  background-color: #fff3c4;
}

/* Float: my gallery's figures side by side, each picture the same size; every section starts below them */
.gallery figure {
  float: left;
  width: 170px;
  margin: 0 12px 12px 0;
}
.gallery img {
  width: 160px;
  height: 110px;
}
.gallery figcaption {
  text-align: center;
}
h2 {
  clear: both;
}

/* Hover and focus: my links turn light yellow under the mouse and the keyboard alike; the keyboard also gets an outline (8.8:1) */
a:hover, a:focus {
  background-color: #fff3c4;
}
a:focus-visible {
  outline: 3px solid #8a1c55;
  outline-offset: 2px;
}

/* A pseudo-class: my first skill in bold. A pseudo-element: a star before each section heading (decoration only) */
.skills li:first-child {
  font-weight: bold;
}
h2::before {
  content: "★ ";
}

/* Flexbox: my skills in a column, with one gap instead of a margin on each item */
.skills {
  display: flex;
  flex-direction: column;
  gap: 0.5em;
}

/* Flexbox instead of float: my gallery figures side by side, wrapping when they do not fit */
.gallery {
  display: flex;
  flex-wrap: wrap;
  gap: 12px;
}

/* Flexbox for my nav bar: a row of items, with gap between them */
nav ul {
  display: flex;
  flex-direction: row;
  gap: 0.5em;
}

/* Media query: at 900px and narrower, my nav items stand one under the other */
@media (max-width: 900px) {
  nav ul {
    flex-direction: column;
  }
}
